;;; persp unread marks from ghostel notifications -*- lexical-binding: t; -*-

(defvar my/persp-unread (make-hash-table :test #'equal)
  "Perspective names with unseen ghostel notifications")

(defun my/persp-unread-p (name)
  (gethash name my/persp-unread))

(defun my/persp-clear-unread (&optional name)
  (remhash (or name (persp-current-name)) my/persp-unread)
  (persp-update-modestring))

(defun my/persps-containing-buffer (buf)
  (let (names)
    (maphash
     (lambda (name persp)
       (when (memq buf (persp-buffers persp))
	 (push name names)))
     (perspectives-hash))
    names))

(defun my/ghostel-notify-and-mark (title body)
  (ghostel-default-notify title body)
  (let ((buf (current-buffer))
	(cur (persp-current-name)))
    (dolist (name (my/persps-containing-buffer buf))
      (unless (equal name cur)
	(puthash name t my/persp-unread)))
    (persp-update-modestring)))

(setq ghostel-notification-function #'my/ghostel-notify-and-mark)
(add-hook 'persp-activated-hook #'my/persp-clear-unread)

;; mode-line mark
(defun my/persp-format-name-unread (orig name)
  (let ((s (funcall orig name)))
    (if (my/persp-unread-p name)
	(concat s (propertize "*" 'face 'warning))
      s)))
;; advice-add 包裝原有的 function 執行
(advice-add 'persp-format-name :around #'my/persp-format-name-unread)

(defun my/persp-switch (name)
  "Switch perspective; list shows * on unread"
  (interactive
   (list
    (completing-read
     "Perspective: "
     (lambda (str pred action)
       (if (eq action 'metadata)
	   '(metadata
	     (annotation-function
	      . (lambda (c)
		  (if (my/persp-unread-p c)
		      (propertize " *" 'face 'warning)
		    "")))
	     (category . perspective))
	 (complete-with-action action
			       (let ((last (and (persp-last) (persp-name (persp-last)))))
				 (if last
				     (cons last (delete last (persp-names)))
				   (persp-names)))
			       str pred)))
     nil nil nil nil
     (let ((last (persp-last)))
       (if last
	   (persp-name last)
	 (persp-current-name))))))
  (persp-switch name))

(with-eval-after-load 'perspective
  (define-key perspective-map (kbd "s") #'my/persp-switch))
